let DIL = "en";
let COGUL = null, SAYIB = null, SAATB = null, TARIHB = null, GORECELI = null;

/* "pt_BR", "zh-Hant-TW", "de-AT" gibi her etiketi elimizdeki dillerden
   birine indirger; karşılığı yoksa null. */
function dilEsle(ham) {
  if (!ham) return null;
  const s = String(ham).replace(/_/g, "-");
  if (SOZLUK[s]) return s;
  const p = s.toLowerCase().split("-"), ana = p[0];
  if (ana === "zh") {
    /* Çince'de belirleyici olan bölge değil yazı sistemi. */
    if (p.indexOf("hant") >= 0 || p.indexOf("tw") >= 0 || p.indexOf("hk") >= 0 || p.indexOf("mo") >= 0) return "zh-Hant";
    return "zh-Hans";
  }
  if (ana === "pt") return "pt-BR";
  for (const k in SOZLUK) if (k.toLowerCase().split("-")[0] === ana) return k;
  return null;
}

function dilSec(istek) {
  if (istek && istek !== "auto") { const k = dilEsle(istek); if (k) return k; }
  const liste = (navigator.languages && navigator.languages.length)
    ? navigator.languages : [navigator.language];
  for (const x of liste) { const k = dilEsle(x); if (k) return k; }
  return "en";
}

/* Biçimlendiriciler dil başına bir kez kurulur — her çağrıda Intl nesnesi
   yaratmak sayaç ekranında saniyede bir dönen döngüde pahalıya geliyor.
   Rakamlar her dilde latin tutuluyor: kadrandaki saat ASCII, karışığı çirkin. */
function dilKur() {
  DIL = dilSec(S && S.settings ? S.settings.lang : "auto");
  const b = DIL + "-u-nu-latn";
  try { COGUL = new Intl.PluralRules(DIL); } catch (e) { COGUL = null; }
  try { SAYIB = new Intl.NumberFormat(b, { maximumFractionDigits: 1 }); } catch (e) { SAYIB = null; }
  try { SAATB = new Intl.DateTimeFormat(b, { hour: "numeric", minute: "2-digit" }); } catch (e) { SAATB = null; }
  try { TARIHB = new Intl.DateTimeFormat(b, { day: "numeric", month: "long", weekday: "long" }); } catch (e) { TARIHB = null; }
  try { GORECELI = new Intl.RelativeTimeFormat(DIL, { numeric: "auto" }); } catch (e) { GORECELI = null; }
}

function soz(anahtar, p) {
  let v = SOZLUK[DIL] && SOZLUK[DIL][anahtar];
  if (v === undefined) v = SOZLUK.en[anahtar];
  if (v === undefined) v = SOZLUK.tr[anahtar];
  if (v === undefined) return anahtar;
  if (typeof v === "object") {
    let kat = "other";
    try { kat = COGUL.select(Number(p && p.n)); } catch (e) {}
    v = v[kat] || v.other;
  }
  if (!p) return v;
  return v.replace(/\{(\w+)\}/g, (m, a) => (p[a] === undefined ? m : p[a]));
}

const pad = n => String(n).padStart(2, "0");
function dayKey(ts) { const d = new Date(ts); return d.getFullYear() + "-" + pad(d.getMonth() + 1) + "-" + pad(d.getDate()); }
function todayKey() { return dayKey(now()); }
/* Ham 24 saat biçimi: <input type="time"> ve dosya adları bunu ister. */
function hhmmHam(ts) { const d = new Date(ts); return pad(d.getHours()) + ":" + pad(d.getMinutes()); }
/* Ekranda görünen saat: dilin kendi biçimi (12 ya da 24 saat). */
function hhmm(ts) { return SAATB ? SAATB.format(new Date(ts)) : hhmmHam(ts); }
/* Sağdan-sola dilde "3:31 م" gibi bir damga, cümlenin geri kalanıyla
   karışıp parçalanıyor. Her damga kendi yalıtımına alınır. */
function saatY(ts) { return yon(hhmm(ts)); }
/* İki saat arası: uçlar da, ikisinin sırası da yerinde kalsın diye
   iç içe iki yalıtım gerekiyor. Soldan-sağda hiçbir işaret eklenmez. */
function aralik(a, b) { return yon(saatY(a) + " – " + saatY(b)); }
function clockStr(ms) {
  if (ms < 0) ms = 0;
  const t = Math.floor(ms / 1000), h = Math.floor(t / 3600), m = Math.floor(t % 3600 / 60), s = t % 60;
  return h > 0 ? h + ":" + pad(m) + ":" + pad(s) : pad(m) + ":" + pad(s);
}
function human(ms) {
  const dk = Math.round(ms / 60000), h = Math.floor(dk / 60), m = dk % 60;
  if (h && m) return soz("sure_sd", { s: h, d: m });
  if (h) return soz("sure_s", { s: h });
  return soz("sure_d", { d: m });
}
function fullDate(k) {
  const [Y, M, D] = k.split("-").map(Number);
  const t = new Date(Y, M - 1, D);
  if (!TARIHB) return k;
  const s = TARIHB.format(t);
  /* Türkçe tasarım dili küçük harf; Intl büyük harfle veriyor. Diğer
     dillerde büyük harf gerekli (Almanca'da ad, İngilizce'de gün adı). */
  return DIL === "tr" ? s.toLocaleLowerCase("tr") : s;
}
function dayLabel(k) {
  if (GORECELI) {
    if (k === todayKey()) return GORECELI.format(0, "day");
    if (k === dayKey(now() - 864e5)) return GORECELI.format(-1, "day");
  }
  return fullDate(k);
}
function timeToTs(v, near) {
  if (!v) return near;
  const [h, m] = v.split(":").map(Number);
  const d = new Date(near); d.setHours(h, m, 0, 0);
  let t = d.getTime();
  if (t < near - 12 * 36e5) t += 864e5;
  return t;
}
const esc = t => String(t).replace(/[<>&]/g, c => ({ "<": "&lt;", ">": "&gt;", "&": "&amp;" }[c]));

/* ---------------- seans hesabı (tek doğruluk kaynağı) ---------------- */
function stats(se, ref) {
  const end = se.endedAt || ref || now();
  let brk = 0, bs = null, breaks = 0, legacy = 0, ping = null;
  for (const e of se.events) {
    if (e.type === "break_start") { bs = e.at; breaks++; }
    else if (e.type === "break_end" && bs != null) { brk += e.at - bs; bs = null; }
    else if (e.type === "presence_ping") { if (ping == null) ping = e.at; }        // eski sürüm verisi
    else if (e.type === "presence_ack") { if (ping != null) { legacy += e.at - ping; ping = null; } }
  }
  if (bs != null) brk += end - bs;
  if (ping != null) legacy += end - ping;
  const gross = end - se.startedAt;
  const net = Math.max(0, gross - brk);
  let un = legacy + (se.awayMs || 0);
  if (se.unconfirmedOk) un = 0;
  un = Math.min(un, net);
  if (un < 60000) un = 0;
  return {
    gross, brk, net, breaks,
    unconfirmed: un, confirmed: Math.max(0, net - un),
    onBreak: bs != null, breakStart: bs, end
  };
}
const activeSession = () => S.active ? S.sessions.find(x => x.id === S.active) : null;
function push(se, type, at) { se.events.push({ type, at: at || now() }); }

/* senkron damgası: her değişiklikte kaydın yaşı güncellenir.
   Birleştirmede "aynı id iki tarafta farklıysa yeni olan kazanır" kuralı buna dayanır. */
function touch(x) { if (x) x.editedAt = now(); return x; }
/* silinen kayıt mezar taşı bırakır, yoksa diğer cihazdan geri dirilir */
function tomb(kind, id) {
  if (!Array.isArray(S.tombstones)) S.tombstones = [];
  const i = S.tombstones.findIndex(t => t.id === id);
  if (i >= 0) S.tombstones[i].at = now();
  else S.tombstones.push({ kind, id, at: now() });
}

/* seansı çalışma/mola parçalarına ayır (kadran için) */
function segments(se, ref) {
  const end = se.endedAt || ref || now();
  const out = []; let cur = { type: "work", from: se.startedAt };
  for (const e of se.events) {
    if (e.type === "break_start" && cur.type === "work") { out.push({ type: "work", from: cur.from, to: e.at }); cur = { type: "break", from: e.at }; }
    else if (e.type === "break_end" && cur.type === "break") { out.push({ type: "break", from: cur.from, to: e.at }); cur = { type: "work", from: e.at }; }
  }
  out.push({ type: cur.type, from: cur.from, to: end });
  return out;
}

    akisV: AKIS_GUNCEL

function takeBreak() {
  const se = activeSession(); if (!se) return;
  if (stats(se).onBreak) return;
  push(se, "break_start"); touch(se); buzz(18); save(); render();
}
function resume() {
  const se = activeSession(); if (!se) return;
  const st = stats(se); if (!st.onBreak) return;
  const len = now() - st.breakStart;
  const ask = S.settings.longBreakAsk;
  if (ask > 0 && len > ask * 60000) { askLongBreak(se, st.breakStart, len); return; }
  doResume(se);
}
function doResume(se, at) {
  push(se, "break_end", at); touch(se);
  lastMilestone = Math.floor(stats(se).net / 60000 / (S.settings.milestone || 1e9));
  buzz(18); save(); render();
}
function finish(at) {
  const se = activeSession(); if (!se) return;
  if (stats(se).onBreak) push(se, "break_end", at);
  push(se, "end", at);
  se.endedAt = at || now(); touch(se);
  S.active = null; releaseWake();
  save(); render();
  const st = stats(se);
  const thr = S.settings.longSessionAsk;
  if (thr > 0 && st.net >= thr * 60000) askAway(se);
  else showSummary(se);
  setTimeout(() => autoSync(true), 1200);
}

/* ---------------- bildirim çubuğundaki canlı sayaç ----------------
   Bildirim yalnızca bir GÖSTERGE. Süre hâlâ olay damgalarından hesaplanıyor;
   burada hiçbir şey sayılmıyor, saklanmıyor. Android kronometreyi kendisi
   işlettiği için sadece metin değiştiğinde haber veriyoruz. */
let nativeSonImza = "";
function nativeBildir(zorla) {
  if (!NATIVE) return;
  const se = activeSession();
  if (!se) {
    if (nativeSonImza !== "yok") { try { NATIVE.gizle(); } catch (e) {} nativeSonImza = "yok"; }
    return;
  }
  const st = stats(se);
  let baslik, alt, temel, geri = false;

  if (st.onBreak) {
    baslik = soz("durum_mola");
    alt = soz("capt_donduruldu", { s: human(st.net) });
    temel = st.breakStart;
  } else if (se.mode === "pomodoro" && se.target > 0) {
    const kalan = se.target * 60000 - st.net;
    baslik = kalan >= 0 ? soz("durum_calisiyor") : soz("bld_hedef_doldu");
    alt = soz("capt_centik", { a: Math.floor(st.net / 60000), b: se.target });
    temel = now() + kalan;          // kalan<0 ise geçmiş bir an: yukarı sayar
    geri = kalan >= 0;
  } else {
    baslik = soz("durum_calisiyor");
    alt = soz("alt_basladin", { t: saatY(se.startedAt) }) + (st.brk ? " · " + soz("ek_mola", { s: human(st.brk) }) : "");
    temel = se.startedAt;
  }

  const imza = baslik + "|" + alt + "|" + Math.round(temel / 1000) + "|" + geri;
  if (!zorla && imza === nativeSonImza) return;
  nativeSonImza = imza;
  try { NATIVE.goster(baslik, alt, String(temel), geri, st.onBreak); } catch (e) {}
}
window.nativeBildir = nativeBildir;

/* Bildirimdeki tuşlardan gelen komutlar. Tek giriş noktası; buradan sonrası

  nativeBildir(true);
};

const NS = "http://www.w3.org/2000/svg";
const CX = 155, CY = 155, R0 = 114, GAP = 16, RMAX = 140;

function mk(tag, a) { const e = document.createElementNS(NS, tag); for (const k in a) e.setAttribute(k, a[k]); return e; }
function arcPath(r, a0, a1) {
  const p = a => [CX + r * Math.cos(a * Math.PI / 180), CY + r * Math.sin(a * Math.PI / 180)];
  const [x0, y0] = p(a0), [x1, y1] = p(a1);
  return `M ${x0.toFixed(2)} ${y0.toFixed(2)} A ${r} ${r} 0 ${Math.abs(a1 - a0) > 180 ? 1 : 0} 1 ${x1.toFixed(2)} ${y1.toFixed(2)}`;
}
function css(n) { return getComputedStyle(document.documentElement).getPropertyValue(n).trim(); }
function notch(g, r, angDeg, h, col, w, op) {
  const a = angDeg * Math.PI / 180;
  g.appendChild(mk("line", {
    x1: (CX + (r - h / 2) * Math.cos(a)).toFixed(2), y1: (CY + (r - h / 2) * Math.sin(a)).toFixed(2),
    x2: (CX + (r + h / 2) * Math.cos(a)).toFixed(2), y2: (CY + (r + h / 2) * Math.sin(a)).toFixed(2),
    stroke: col, "stroke-width": w, "stroke-linecap": "round", opacity: op == null ? 1 : op
  }));
}

/* kronometre: dakikada bir çentik, 60 çentikte bir halka */
function drawStopwatchDial(svg, se, isStatic) {
  svg.innerHTML = "";
  const segs = segments(se);
  const start = se.startedAt;
  const T = ((se.endedAt || now()) - start) / 60000;
  const laps = Math.max(1, Math.ceil(T / 60));
  const s = Math.min(1, RMAX / (R0 + (laps - 1) * GAP));
  const g = mk("g", { transform: `translate(${CX} ${CY}) scale(${s.toFixed(4)}) translate(${-CX} ${-CY})` });
  svg.appendChild(g);

  const cAcc = css("--accent"), cTrk = css("--track"), cCool = css("--cool");

  for (let k = 0; k < laps; k++) {
    const from = k * 60, to = Math.min(T, (k + 1) * 60);
    if (to <= from) continue;
    const a0 = (from - k * 60) * 6 - 90, a1 = (to - k * 60) * 6 - 90;
    g.appendChild(mk("path", { d: arcPath(R0 + k * GAP, a0, Math.max(a1, a0 + .01)),
      fill: "none", stroke: cTrk, "stroke-width": 1.2, "stroke-linecap": "round" }));
  }

  const kindAt = m => {
    const ts = start + m * 60000;
    for (const sg of segs) if (ts >= sg.from && ts < sg.to) return sg.type;
    return "work";
  };
  const full = Math.floor(T);
  for (let m = 0; m < full; m++) {
    if (kindAt(m + .5) === "break") continue;
    const k = Math.floor(m / 60), mm = m % 60;
    let h = 6.5;
    if (mm === 0) h = 14; else if (mm % 15 === 0) h = 12; else if (mm % 5 === 0) h = 9;
    notch(g, R0 + k * GAP, mm * 6 - 90, h, cAcc, 1.8, 1);
  }

  for (const sg of segs) {
    if (sg.type !== "break") continue;
    let a = (sg.from - start) / 60000, b = Math.min((sg.to - start) / 60000, T);
    while (a < b) {
      const k = Math.floor(a / 60), end = Math.min(b, (k + 1) * 60);
      const a0 = (a - k * 60) * 6 - 90, a1 = (end - k * 60) * 6 - 90;
      if (a1 > a0 + .05) g.appendChild(mk("path", { d: arcPath(R0 + k * GAP, a0, a1),
        fill: "none", stroke: cCool, "stroke-width": 2.6, "stroke-linecap": "round", opacity: .5 }));
      a = end;
    }
  }

  if (T > 0 && !isStatic) {
    const k = Math.floor(T / 60), r = R0 + k * GAP, a = ((T % 60) * 6 - 90) * Math.PI / 180;
    const d = mk("circle", { cx: (CX + r * Math.cos(a)).toFixed(2), cy: (CY + r * Math.sin(a)).toFixed(2), r: 3.4, fill: cAcc });
    d.setAttribute("class", "head-dot");
    g.appendChild(d);
  }
}

/* pomodoro: hedef kadar çentik, teker teker kazınır */
function drawPomodoroDial(svg, target, doneMin) {
  svg.innerHTML = "";
  const g = mk("g", {});
  svg.appendChild(g);
  const cAcc = css("--accent"), cTrk = css("--track");
  const r = 118, n = Math.max(1, target);
  g.appendChild(mk("circle", { cx: CX, cy: CY, r: r, fill: "none", stroke: cTrk, "stroke-width": 1.2 }));
  const done = Math.max(0, Math.min(n, Math.floor(doneMin)));
  for (let i = 0; i < n; i++) {
    const carved = i < done;
    notch(g, r, i * (360 / n) - 90, (i % 5 === 0) ? 11 : 7, carved ? cAcc : cTrk, carved ? 1.9 : 1.4, 1);
  }
  if (done > 0 && done < n) {
    const a = (done * (360 / n) - 90) * Math.PI / 180;
    const d = mk("circle", { cx: (CX + r * Math.cos(a)).toFixed(2), cy: (CY + r * Math.sin(a)).toFixed(2), r: 3.4, fill: cAcc });
    d.setAttribute("class", "head-dot");
    g.appendChild(d);
  }
}

/* ============================================================
   TOHUMLU RASTGELELİK
   Aynı seans her açılışta aynı deseni verir. Kayıt görünürken
   değişmemeli — desen süslemeden çok parmak izi gibi çalışıyor.
   ============================================================ */
function hash32(t) { let h = 2166136261; for (let i = 0; i < t.length; i++) { h ^= t.charCodeAt(i); h = Math.imul(h, 16777619); } return h >>> 0; }
function rng32(a) {
  a = a >>> 0;
  return () => { a = a + 0x6D2B79F5 | 0; let t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; };
}

/* ============================================================
   ÇUBUK — seansın kaydı, çentikten
   Dakikada bir kesik, her saat yeni bir sütun. Mola kesik bırakmaz,
   kenarda serin bir iz bırakır. Sayılmayan süre nerede olduğu
   bilinmediği için çubuğa yazılmaz; sağda kesikli bir ray olarak
   yalnız MİKTAR olarak durur. Konum uydurmuyoruz.
   ============================================================ */

/* app seansı → çubuk verisi (dakika cinsinden, seans başına göre) */
function cubukVeri(se, ref) {
  const st = stats(se, ref);
  const end = se.endedAt || ref || now();
  const dk = Math.max(1, Math.round((end - se.startedAt) / 60000));
  const molalar = [];
  for (const sg of segments(se, ref)) {
    if (sg.type !== "break") continue;
    const a = (sg.from - se.startedAt) / 60000, b = (sg.to - se.startedAt) / 60000;
    if (b > a + .01) molalar.push([Math.max(0, a), Math.min(dk, b)]);
  }
  return { id: se.id, dk, molalar, bos: [], sayilmayan: Math.round(st.unconfirmed / 60000) };
}

/* bir günün bütün seansları tek çubukta: arka arkaya, aralarında bir boşluk */
function gunCubuk(list, key) {
  const sirali = list.slice().sort((a, b) => a.startedAt - b.startedAt);
  const molalar = [], bos = [];
  let m = 0, sayilmayan = 0, id = key || "";
  sirali.forEach((se, i) => {
    if (i > 0) { bos.push([m, m + 1]); m += 1; }
    const v = cubukVeri(se);
    for (const b of v.molalar) molalar.push([m + b[0], m + b[1]]);
    sayilmayan += v.sayilmayan;
    m += v.dk;
    id += se.id;
  });
  return { id, dk: Math.max(1, m), molalar, bos, sayilmayan };
}

/* v = { id, dk, molalar, bos, sayilmayan }  — hepsi dakika */
function cizCubuk(svg, v, o) {
  o = o || {};
  const RP = o.rp || 5.4, SW = o.sw || 30, GAP = o.gap || 13, TOP = o.top || 5;
  const reveal = !!o.reveal, sure = o.sureMs || 1500;
  svg.innerHTML = "";
  const dk = Math.max(1, Math.round(v.dk));
  const kolon = Math.max(1, Math.ceil(dk / 60));
  const satir = kolon > 1 ? 60 : dk;
  const W = kolon * (SW + GAP) - GAP, H = TOP * 2 + Math.max(1, satir - 1) * RP;
  const SAG = (v.sayilmayan > 0 ? 9 : 2);
  svg.setAttribute("viewBox", `-7 0 ${(W + 7 + SAG).toFixed(1)} ${H.toFixed(1)}`);
  svg.setAttribute("width", (W + 7 + SAG).toFixed(1));
  svg.setAttribute("height", H.toFixed(1));

  const acc = css("--accent"), cool = css("--cool"), ink = css("--ink"), line = css("--line");
  const R = rng32(hash32(v.id || "x"));
  const j = []; for (let i = 0; i < dk * 3 + 40; i++) j.push(R());   // sabit titreşim havuzu
  const tur = m => {
    for (const b of (v.bos || [])) if (m >= b[0] && m < b[1]) return "bos";
    for (const b of (v.molalar || [])) if (m >= b[0] && m < b[1]) return "mola";
    return "is";
  };

  const g = mk("g", {}); svg.appendChild(g);
  const colX = k => k * (SW + GAP);
  const yOf = m => TOP + (m % 60) * RP;

  /* sırt — çubuğun kendisi */
  for (let k = 0; k < kolon; k++) {
    const son = Math.min(60, dk - k * 60);
    const l = mk("line", { x1: colX(k), y1: TOP - 2.5, x2: colX(k), y2: yOf(son - 1) + 2.5,
      stroke: line, "stroke-width": 1.1, "stroke-linecap": "round" });
    if (reveal) { l.style.opacity = 0; l.style.transition = "opacity .5s ease";
      setTimeout(() => l.style.opacity = 1, 60 + k * 90); }
    g.appendChild(l);
  }

  /* mola — kâğıtta boşluk, kenarda serin bir iz */
  for (const b of (v.molalar || [])) {
    let a = b[0];
    while (a < Math.min(b[1], dk)) {
      const k = Math.floor(a / 60), son = Math.min(b[1], dk, (k + 1) * 60);
      if (son > a) {
        const l = mk("line", { x1: colX(k) - 4, y1: yOf(a) - RP * .4, x2: colX(k) - 4, y2: yOf(son - 1) + RP * .4,
          stroke: cool, "stroke-width": Math.max(1.6, SW * .08), "stroke-linecap": "round", opacity: .45 });
        if (reveal) { l.style.opacity = 0; l.style.transition = "opacity .6s ease";
          setTimeout(() => l.style.opacity = .45, 200 + (a / dk) * sure); }
        g.appendChild(l);
      }
      a = son;
    }
  }

  /* kesikler */
  const adim = Math.min(.02, (sure / 1000) / Math.max(1, dk));
  let n = 0;
  for (let m = 0; m < dk; m++) {
    const t = tur(m);
    if (t !== "is") continue;
    const k = Math.floor(m / 60), mm = m % 60;
    let boy = .54;
    if (mm === 0) boy = 1.02; else if (mm % 15 === 0) boy = .86; else if (mm % 5 === 0) boy = .72;
    const r1 = j[m * 3], r2 = j[m * 3 + 1], r3 = j[m * 3 + 2];
    const L = SW * boy * (1 + (r1 - .5) * .09);
    const x0 = colX(k), y0 = yOf(m) + (r2 - .5) * .7;
    const y1 = y0 + (r3 - .5) * 1.5;                       // kesiğin ucu hafif eğik
    const kal = mm === 0 ? 2.0 : (mm % 5 === 0 ? 1.7 : 1.35 + (r1 - .5) * .35);
    const op = mm === 0 ? 1 : .74 + r3 * .22;

    /* kesiğin gölgesi — kâğıtta derinlik */
    const s = mk("line", { x1: x0 + .6, y1: y0 + 1.1, x2: x0 + L * .92, y2: y1 + 1.1,
      stroke: ink, "stroke-width": .8, "stroke-linecap": "round", opacity: .10, pathLength: 1 });
    if (reveal) { s.setAttribute("class", "carve"); s.style.animationDelay = (.06 + n * adim) + "s"; }
    g.appendChild(s);

    const c = mk("line", { x1: x0, y1: y0, x2: x0 + L, y2: y1, stroke: acc,
      "stroke-width": kal, "stroke-linecap": "round", opacity: op, pathLength: 1 });
    if (reveal) { c.setAttribute("class", "carve"); c.style.animationDelay = (.05 + n * adim) + "s"; }
    g.appendChild(c);
    n++;
  }

  /* sayılmayan süre — miktar, konum değil.
     Son sütunun sağında kesikli bir ray; kaç çentiğin sayılmadığını gösterir. */
  if (v.sayilmayan > 0) {
    const k = kolon - 1, boyu = Math.min(v.sayilmayan, satir);
    const x = colX(k) + SW + 5;
    const l = mk("line", { x1: x, y1: TOP, x2: x, y2: TOP + Math.max(1, boyu - 1) * RP,
      stroke: acc, "stroke-width": 1.6, "stroke-linecap": "round",
      "stroke-dasharray": (RP * .55).toFixed(2) + " " + (RP * .45).toFixed(2), opacity: .42 });
    if (reveal) { l.style.opacity = 0; l.style.transition = "opacity .5s ease";
      setTimeout(() => l.style.opacity = .42, 120 + n * adim * 1000); }
    g.appendChild(l);
  }
  return (.05 + n * adim) * 1000 + 420;
}

/* Akış'ın yönleri.

   katman: [pay, kalınlık, karanlık α, açık α, yıkama kalınlığı]
   Paylar 1'e tamamlanır. Bir şeridin hangi katmana düştüğü rastgele değil,
   alanın kendi gürültüsünden geliyor (bkz. akisSira): kalın damarlar bölge
   bölge kümeleniyor, ebru kâğıdındaki gibi.

   ton/isik: şeridin ağırlığına göre renk kayması. Kalın damar biraz daha
   koyu ve derin, kılcal tüy biraz daha açık. Yanardönerlik buradan geliyor.

   alan: şeritlerin geometrisi.
     dsep  şerit aralığı — komşuluk taraması 3×3 olduğu için gerçek aralık ~3×dsep
     maxl  bir şeridin en fazla uzunluğu
     dene  kaç başlangıç noktası denenecek (maliyetin çoğu burada)
     af    açı alanının üç oktavı: sayı büyüdükçe frekans artar, desen kıvrılır.
           Eski değer (1 ve 3) neredeyse düz bir alan veriyordu — ekranı boş
           gösteren asıl sebep buydu, saydamlık değil.
     of    doluş sırasının frekansı, jit  o sıraya karışan rastgelelik.
           Düşük of = alan tek bir kütle hâlinde büyür ve seansın yarısında
           ekranın yarısı boş kalır. Yüksek of + biraz jit = birkaç yerden
           birden büyür, her an dolu görünür. */

/* Seansın akisV damgası → hangi yönle çizilecek. Yeni bir yöne geçilirse
   buraya 3 eklenir ve AKIS_GUNCEL 3 olur; eskiler eski hâllerini korur. */
const AKIS_SURUM = { 1: "sade", 2: "ebru" };
const AKIS_GUNCEL = 2;
function akisYonu(se) { return AKIS_SURUM[se && se.akisV] || AKIS_SURUM[1]; }

/* #RRGGBB → [h,s,l]. Palet değişkenleri hep hex; değilse null döner ve
   renk olduğu gibi kullanılır. */
function hexHsl(x) {
  const m = /^#([0-9a-f]{6})$/i.exec(x || ""); if (!m) return null;
  const n = parseInt(m[1], 16), r = (n >> 16) / 255, g = (n >> 8 & 255) / 255, b = (n & 255) / 255;
  const mx = Math.max(r, g, b), mn = Math.min(r, g, b), l = (mx + mn) / 2, d = mx - mn;
  if (!d) return [0, 0, l];
  const s = l > .5 ? d / (2 - mx - mn) : d / (mx + mn);
  let h = mx === r ? (g - b) / d + (g < b ? 6 : 0) : mx === g ? (b - r) / d + 2 : (r - g) / d + 4;
  return [h * 60, s, l];
}

function akisRenk(hsl, taban, q, M) {
  if (!hsl || !M.ton) return taban;
  const k = q * 2 - 1;                       /* -1 tüy … +1 damar */
  const h = (hsl[0] + k * M.ton + 360) % 360;
  const l = Math.min(.94, Math.max(.06, hsl[2] - k * M.isik));
  return "hsl(" + h.toFixed(1) + "," + (hsl[1] * 100).toFixed(1) + "%," + (l * 100).toFixed(1) + "%)";
}

function h2i(i, j, s) {
  let n = Math.imul(i | 0, 374761393) ^ Math.imul(j | 0, 668265263) ^ Math.imul(s | 0, 362437);
  n = Math.imul(n ^ (n >>> 13), 1274126177); return ((n ^ (n >>> 16)) >>> 0) / 4294967296;
}
function vnoise(x, y, s) {
  const i = Math.floor(x), j = Math.floor(y), fx = x - i, fy = y - j;
  const u = fx * fx * (3 - 2 * fx), v = fy * fy * (3 - 2 * fy);
  const a = h2i(i, j, s), b = h2i(i + 1, j, s), c = h2i(i, j + 1, s), d = h2i(i + 1, j + 1, s);
  return (a * (1 - u) + b * u) * (1 - v) + (c * (1 - u) + d * u) * v;
}
function fbm(x, y, s) { return vnoise(x, y, s) * .6 + vnoise(x * 2.1, y * 2.1, s + 7) * .27 + vnoise(x * 4.3, y * 4.3, s + 19) * .13; }

/* akış çizgilerini bir kere hesapla; sonra sadece daha fazlası çizilir */
function akisAlan(seed, W, H, A) {
  const R = rng32(seed);
  const DSEP = A.dsep, STEP = 2.0, MAXL = A.maxl;
  const gw = Math.ceil(W / DSEP) + 2, gh = Math.ceil(H / DSEP) + 2;
  const dolu = new Uint8Array(gw * gh);
  const f = A.af;
  const aci = (x, y) => (fbm(x / (W / f[0]), y / (W / f[0]), seed) * 2 - 1) * Math.PI * 1.9
                      + (fbm(x / (W / f[1]), y / (W / f[1]), seed + 3) - .5) * 1.1
                      + (f[2] ? (fbm(x / (W / f[2]), y / (W / f[2]), seed + 11) - .5) * .5 : 0);
  const bosMu = (x, y) => {
    const gi = (x / DSEP | 0) + 1, gj = (y / DSEP | 0) + 1;
    if (gi < 0 || gj < 0 || gi >= gw || gj >= gh) return false;
    for (let a = -1; a <= 1; a++) for (let b = -1; b <= 1; b++) {
      const q = (gj + b) * gw + (gi + a); if (q >= 0 && q < dolu.length && dolu[q]) return false;
    }
    return true;
  };
  const isaretle = (x, y) => { const gi = (x / DSEP | 0) + 1, gj = (y / DSEP | 0) + 1;
    if (gi >= 0 && gj >= 0 && gi < gw && gj < gh) dolu[gj * gw + gi] = 1; };

  /* Doluş sırası: alan tek kütle hâlinde değil, birkaç yerden birden büyüsün. */
  const denemeler = [];
  for (let i = 0; i < A.dene; i++) {
    const x = R() * W, y = R() * H;
    denemeler.push({ x, y, o: fbm(x / (W / A.of), y / (W / A.of), seed + 41) * (1 - A.jit) + R() * A.jit });
  }
  denemeler.sort((a, b) => a.o - b.o);

  const seritler = [];
  for (const d of denemeler) {
    if (!bosMu(d.x, d.y)) continue;
    let pts = [];
    for (const yon of [1, -1]) {
      let x = d.x, y = d.y; const par = [];
      for (let k = 0; k < MAXL / STEP; k++) {
        const a = aci(x, y);
        x += Math.cos(a) * STEP * yon; y += Math.sin(a) * STEP * yon;
        if (x < -20 || y < -20 || x > W + 20 || y > H + 20) break;
        if (k > 2 && !bosMu(x, y)) break;
        par.push(x, y);
      }
      if (yon === 1) pts.push(...par);
      else { const rev = []; for (let k = par.length - 2; k >= 0; k -= 2) rev.push(par[k], par[k + 1]); pts.unshift(...rev); }
    }
    if (pts.length < 14) continue;
    for (let k = 0; k < pts.length; k += 2) isaretle(pts[k], pts[k + 1]);
    seritler.push(Float32Array.from(pts));
  }
  return seritler;
}

/* Her şerit için 0..1 arası bir "ağırlık sırası". Şeridin orta noktasındaki
   gürültüden çıkıyor, sonra sıralanıp yüzdeye çevriliyor — böylece katman
   payları (%62 tüy, %28 orta, %10 damar) her seansta birebir tutuyor ve
   kalın damarlar rastgele serpilmek yerine bölge bölge kümeleniyor. */
function akisSira(seritler, W, seed) {
  const n = seritler.length, w = new Float32Array(n), idx = new Int32Array(n);
  for (let i = 0; i < n; i++) {
    const p = seritler[i], k = (p.length >> 2) << 1;
    w[i] = fbm(p[k] / W * 1.7, p[k + 1] / W * 1.7, seed + 91);
    idx[i] = i;
  }
  const sirali = Array.from(idx).sort((a, b) => w[a] - w[b]);
  const s = new Float32Array(n);
  for (let r = 0; r < n; r++) s[sirali[r]] = n > 1 ? r / (n - 1) : 0;
  return s;
}

/* Tuvalleri ekran ölçüsüne göre kurar. Hem seans akışı hem ilk açılış
   tanıtımı buradan geçiyor. */
function akisTuval() {
  const cv = el("akis"); if (!cv) return null;
  AK.dpr = Math.min(1.75, window.devicePixelRatio || 1);
  AK.w = 300;
  AK.h = Math.max(300, Math.round(300 * innerHeight / Math.max(1, innerWidth)));
  AK.pw = Math.round(AK.w * AK.dpr); AK.ph = Math.round(AK.h * AK.dpr);
  cv.width = AK.pw; cv.height = AK.ph;
  AK.ctx = cv.getContext("2d");
  AK.off = document.createElement("canvas");
  AK.off.width = AK.pw; AK.off.height = AK.ph;
  AK.offCtx = AK.off.getContext("2d");
  AK.offCtx.setTransform(AK.dpr, 0, 0, AK.dpr, 0, 0);
  return cv;
}

function akisBaslat(se) {
  const cv = akisTuval(); if (!cv) return;
  AK.sid = se.id; AK.hazir = false; AK.ciz = 0; AK.lastMin = -1; AK.serit = [];
  const id = AK.sid;
  /* hesap ağır; önce arayüz boyansın diye bir tık sonraya bırakılıyor */
  setTimeout(() => {
    if (AK.sid !== id) return;
    /* duvar saati değil, tek yönlü sayaç: simülatör hızlandırsa bile doğru ölçer */
    const t0 = performance.now();
    AK.tohum = hash32(id);
    AK.alanYon = AKSEC;
    AK.serit = akisAlan(AK.tohum, AK.w, AK.h, (AKMOD[AKSEC] || AKMOD.sade).alan);
    AK.sira = akisSira(AK.serit, AK.w, AK.tohum);
    /* çok yavaş cihazda desenden vazgeç — uygulamanın geri kalanı aynen çalışır */
    if (performance.now() - t0 > 1400) { AK.kapali = true; AK.serit = []; AK.sira = null; el("akis").classList.remove("on"); return; }
    AK.hazir = true; akisIsle();
  }, 60);
}

/* [bas, son) aralığındaki şeritleri K.g bağlamına ekler.
   K = { g, serit, sira, mod }; molaMi(i) o şeridin mola dakikasına düşüp
   düşmediğini söyler. Hem canlı akış hem kayıt resmi buradan geçiyor. */
function akisCiz(K, bas, son, molaMi) {
  const g = K.g; if (!g) return;
  const M = AKMOD[K.mod] || AKMOD.sade;
  const acc = css("--accent"), cool = css("--cool");
  const accH = hexHsl(acc), coolH = hexHsl(cool);
  const koyu = document.documentElement.classList.contains("dark");
  g.lineCap = "round"; g.lineJoin = "round";
  for (let i = bas; i < son; i++) {
    const mola = molaMi(i);

    /* şeridin ağırlığı → hangi katman, hangi ton */
    const q = K.sira ? K.sira[i] : .5;
    let pay = 0, kat = M.katman[0];
    for (const k of M.katman) { pay += k[0]; kat = k; if (q < pay) break; }
    const a = (koyu ? kat[2] : kat[3]) * (mola ? .8 : 1);

    const p = K.serit[i];
    g.beginPath(); g.moveTo(p[0], p[1]);
    for (let k = 2; k < p.length; k += 2) g.lineTo(p[k], p[k + 1]);

    /* yıkama: aynı yol, geniş ve çok soluk. Ebrudaki gölleşmeyi bu veriyor. */
    if (kat[4]) {
      g.strokeStyle = mola ? cool : acc;
      g.lineWidth = kat[4]; g.globalAlpha = a * .13;
      g.stroke();
    }
    g.strokeStyle = akisRenk(mola ? coolH : accH, mola ? cool : acc, q, M);
    g.lineWidth = kat[1] * (mola ? .85 : 1);
    g.globalAlpha = a;
    g.stroke();
  }
  g.globalAlpha = 1;
}

function akisIsle() {
  const se = activeSession();
  if (!se || !AK.hazir || AK.sid !== se.id) return;
  const top = AK.serit.length; if (!top) return;
  const M = AKMOD[AKSEC] || AKMOD.sade;
  const dk = (stats(se).end - se.startedAt) / 60000;
  const hedef = Math.floor(top * Math.min(1, dk / M.dolum));
  if (hedef > AK.ciz) {
    const segs = segments(se);
    akisCiz({ g: AK.offCtx, serit: AK.serit, sira: AK.sira, mod: AKSEC }, AK.ciz, hedef, i => {
      const ts = se.startedAt + (i / top * M.dolum) * 60000;
      for (const sg of segs) if (ts >= sg.from && ts < sg.to) return sg.type === "break";
      return false;
    });
    AK.ciz = hedef;
  }
  akisBirlestir();
}

/* alanı ekrana bas, sonra kadranın çevresini ve alt şeridi aç */
function akisBirlestir() {
  const g = AK.ctx; if (!g) return;
  g.setTransform(1, 0, 0, 1, 0, 0);
  g.clearRect(0, 0, AK.pw, AK.ph);
  g.drawImage(AK.off, 0, 0);
  g.globalCompositeOperation = "destination-out";
  const k = AK.pw / Math.max(1, innerWidth);
  const M = AKMOD[AKSEC] || AKMOD.sade;

  const r = el("dial").getBoundingClientRect();
  if (r.width > 4) {
    const cx = (r.left + r.width / 2) * k, cy = (r.top + r.height / 2) * k;
    const ic = Math.max(1, r.width * M.halo[0] * k), dis = Math.max(ic + 2, r.width * M.halo[1] * k);
    const gr = g.createRadialGradient(cx, cy, ic, cx, cy, dis);
    gr.addColorStop(0, "rgba(0,0,0,1)"); gr.addColorStop(1, "rgba(0,0,0,0)");
    g.fillStyle = gr; g.fillRect(cx - dis, cy - dis, dis * 2, dis * 2);
  }
  const alt = AK.ph * M.alt;
  const gl = g.createLinearGradient(0, AK.ph - alt, 0, AK.ph);
  gl.addColorStop(0, "rgba(0,0,0,0)"); gl.addColorStop(1, "rgba(0,0,0,1)");
  g.fillStyle = gl; g.fillRect(0, AK.ph - alt, AK.pw, alt);

  g.globalCompositeOperation = "source-over";
}

/* ============================================================
   SEANS RESMİ — kayıtta saklanmayan, her seferinde yeniden üretilen görüntü

   Seans bittikten sonra alanının son hâli. Hiçbir dosya tutulmuyor: desen
   seansın kimliğinden türediği için resim her açılışta aynı çıkıyor. Yer
   kaplayan tek şey kullanıcının "kaydet" dediği PNG.
   ============================================================ */

/* İstenen ölçüde bir tuval döndürür. tam=true ise alan seansın gerçek
   süresine göre değil sonuna kadar doldurulur (kullanılmıyor, elde dursun). */
function seansResmi(se, W, H, dpr, imza) {
  const yon = akisYonu(se), M = AKMOD[yon] || AKMOD.sade;
  const cv = document.createElement("canvas");
  cv.width = Math.round(W * dpr); cv.height = Math.round(H * dpr);
  const g = cv.getContext("2d");
  g.setTransform(dpr, 0, 0, dpr, 0, 0);
  g.fillStyle = css("--bg"); g.fillRect(0, 0, W, H);

  const tohum = hash32(se.id);
  const serit = akisAlan(tohum, W, H, M.alan);
  const sira = akisSira(serit, W, tohum);
  const st = stats(se), segs = segments(se);
  const dk = (st.end - se.startedAt) / 60000;
  const top = serit.length;
  const son = Math.floor(top * Math.min(1, dk / M.dolum));
  akisCiz({ g, serit, sira, mod: yon }, 0, son, i => {
    const ts = se.startedAt + (i / top * M.dolum) * 60000;
    for (const sg of segs) if (ts >= sg.from && ts < sg.to) return sg.type === "break";
    return false;
  });

  /* İmza: tarih ve süre. Resim ne olduğunu kendi söylesin. */
  if (imza) {
    const d = new Date(se.startedAt);
    const t1 = d.toLocaleDateString(DIL + "-u-nu-latn", { day: "numeric", month: "long", year: "numeric" });
    const t2 = human(st.confirmed) + (st.brk ? " · " + soz("ek_mola", { s: human(st.brk) }) : "");
    g.globalAlpha = 1;
    g.textAlign = "center";
    g.fillStyle = css("--muted");
    g.font = (W * .038).toFixed(1) + "px -apple-system,system-ui,sans-serif";
    g.fillText(t1, W / 2, H * .79);
    g.fillStyle = css("--faint");
    g.font = (W * .033).toFixed(1) + "px -apple-system,system-ui,sans-serif";
    g.fillText(t2, W / 2, H * .79 + W * .052);
  }
  return cv;
}

function tanitKapat() {
  if (!AK.tanit) return;
  cancelAnimationFrame(tanitKare);
  AK.tanit = false; AK.sid = null; AK.hazir = false; AK.serit = []; AK.sira = null;

function akisTanitim() {
  if (AK.kapali || AK.tanit || activeSession() || tab !== "timer") return;

  const cv = akisTuval(); if (!cv) return;
  AK.tanit = true; AK.sid = "__tanitim__"; AK.hazir = false;

  setTimeout(() => {
    if (!AK.tanit) return;
    const t0 = performance.now();
    AK.tohum = hash32("kertik-tanitim");        /* herkeste aynı desen: tasarlanmış bir an */
    AK.serit = akisAlan(AK.tohum, AK.w, AK.h, (AKMOD[AKSEC] || AKMOD.sade).alan);
    AK.sira = akisSira(AK.serit, AK.w, AK.tohum);
    /* seans akışıyla aynı güvenlik kapısı: yavaş cihazda hiç gösterme */
    if (performance.now() - t0 > 1400 || !AK.serit.length) { AK.kapali = true; tanitKapat(); return; }
    tanitOynat();
  }, 60);
}

function tanitOynat() {
  const top = AK.serit.length, SURE = 3400;
  const cv = el("akis"), yazi = el("tanitYazi");
  cv.classList.add("on");
  const t0 = performance.now();
  let ciz = 0;
  const yok = () => false;
  const TK = () => ({ g: AK.offCtx, serit: AK.serit, sira: AK.sira, mod: AKSEC });
  const adim = () => {
    if (!AK.tanit) return;
    const t = (performance.now() - t0) / SURE;
    if (t >= 1) {
      if (ciz < top) akisCiz(TK(), ciz, top, yok);
      akisBirlestir();
      if (yazi) yazi.classList.remove("on");
      cv.classList.remove("on");                 /* CSS 1.1 sn'de soldurur */
      setTimeout(tanitKapat, 1250);
      return;
    }
    /* hızlı başlar, sona doğru yavaşlar — akıp duruyormuş gibi */
    const hedef = Math.floor(top * (1 - Math.pow(1 - t, 2.4)));
    if (hedef > ciz) { akisCiz(TK(), ciz, hedef, yok); ciz = hedef; akisBirlestir(); }
    if (yazi && t > .3) yazi.classList.add("on");
    tanitKare = requestAnimationFrame(adim);
  };
  tanitKare = requestAnimationFrame(adim);
}

}

/* İki cihazın durumunu birleştirir. Saf fonksiyon: girdileri değiştirmez.
   Kurallar:
     1. Aynı id iki tarafta varsa daha yeni editedAt kazanır.
     2. Silinen kayıt tombstone taşır. Tombstone kaydın son
        düzenlemesinden yeniyse silme kazanır; değilse kayıt geri gelir
        (yani bir cihazda silinip diğerinde düzeltilen kayıt kaybolmaz).
     3. Ayarlar bir bütün olarak son yazan cihazınkidir.
     4. Aktif seans: elindeki cihaz (A) öncelikli. Hiçbir şey uydurulmaz. */
function mergeStates(A, B, opts) {
  opts = opts || {};
  const nowTs = opts.now || Date.now();
  const KEEP_TOMB = 180 * 864e5;   // mezar taşları 6 ay saklanır, sonra temizlenir
  const ADOPT = 12 * 36e5;         // başka cihazın açık seansını en fazla 12 saat sonra devral

  const stampOf = x => x.editedAt || x.closedAt || x.endedAt || x.createdAt || x.startedAt || 0;

  const tombs = new Map();
  for (const t of [].concat(A.tombstones || [], B.tombstones || [])) {
    if (!t || !t.id) continue;
    const p = tombs.get(t.id);
    if (!p || t.at > p.at) tombs.set(t.id, { kind: t.kind, id: t.id, at: t.at });
  }

  function mergeList(a, b) {
    const m = new Map();
    for (const x of a || []) if (x && x.id) m.set(x.id, x);
    for (const x of b || []) {
      if (!x || !x.id) continue;
      const p = m.get(x.id);
      if (!p || stampOf(x) > stampOf(p)) m.set(x.id, x);
    }
    const out = [];
    for (const x of m.values()) {
      const t = tombs.get(x.id);
      if (t && t.at >= stampOf(x)) continue;   // silme kazandı
      if (t) tombs.delete(x.id);               // düzeltme silmeden yeni → kayıt geri geldi
      out.push(x);
    }
    return out;
  }

  const sessions = mergeList(A.sessions, B.sessions).sort((x, y) => x.startedAt - y.startedAt);
  const tasks = mergeList(A.tasks, B.tasks).sort((x, y) => (x.createdAt || 0) - (y.createdAt || 0));

  const aSet = A.settingsEditedAt || 0, bSet = B.settingsEditedAt || 0;
  const settings = (bSet > aSet && B.settings) ? B.settings : (A.settings || B.settings);

  let active = null;
  const openA = A.active ? sessions.find(s => s.id === A.active && !s.endedAt) : null;
  const openB = B.active ? sessions.find(s => s.id === B.active && !s.endedAt) : null;
  if (openA) active = openA.id;
  else if (openB && nowTs - openB.startedAt < ADOPT) active = openB.id;

  const tombstones = Array.from(tombs.values()).filter(t => nowTs - t.at < KEEP_TOMB);

  return { sessions, tasks, settings, active, tombstones, settingsEditedAt: Math.max(aSet, bSet) };
}

/* MainActivity.kt'deki googleYetkiIste()'nin karşılığı. `interaktif=false`

function nativeYetkiIste(interaktif) {
  if (!NATIVE) return Promise.reject(new Error("yerel köprü yok"));
  return new Promise((resolve, reject) => {
    gYetkiBekleyen = { resolve, reject };
    NATIVE.googleYetkiIste(interaktif);
  });
}
window.kertikYetkiSonuc = function (token, hata) {
  const b = gYetkiBekleyen; gYetkiBekleyen = null;
  if (!b) return;
  if (token) { gToken = token; gTokenExp = now() + 55 * 60000; b.resolve(token); }
  else b.reject(new Error(hata || "yetki alınamadı"));
};

async function driveFind(tok) {
  const q = encodeURIComponent("name='" + GOOGLE.FILE + "' and trashed=false");
  const r = await fetch(GOOGLE.FILES + "?spaces=appDataFolder&q=" + q + "&fields=files(id)",
    { headers: { Authorization: "Bearer " + tok } });
  if (!r.ok) throw new Error("drive/list " + r.status);
  const j = await r.json();
  return (j.files && j.files[0] && j.files[0].id) || null;
}
async function driveRead(tok, id) {
  const r = await fetch(GOOGLE.FILES + "/" + id + "?alt=media", { headers: { Authorization: "Bearer " + tok } });
  if (r.status === 404) return null;
  if (!r.ok) throw new Error("drive/read " + r.status);
  try { return await r.json(); } catch (e) { return null; }
}
async function driveWrite(tok, id, data) {
  const body = JSON.stringify(data);
  if (id) {
    const r = await fetch(GOOGLE.UPLOAD + "/" + id + "?uploadType=media&fields=id", {
      method: "PATCH",
      headers: { Authorization: "Bearer " + tok, "Content-Type": "application/json" },
      body
    });
    if (r.ok) return id;
    if (r.status !== 404) throw new Error("drive/write " + r.status);
  }
  const B = "kertikSinir";
  const multipart =
    "--" + B + "\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n" +
    JSON.stringify({ name: GOOGLE.FILE, parents: ["appDataFolder"] }) + "\r\n" +
    "--" + B + "\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n" +
    body + "\r\n--" + B + "--";
  const r2 = await fetch(GOOGLE.UPLOAD + "?uploadType=multipart&fields=id", {
    method: "POST",
    headers: { Authorization: "Bearer " + tok, "Content-Type": 'multipart/related; boundary="' + B + '"' },
    body: multipart
  });
  if (!r2.ok) throw new Error("drive/create " + r2.status);
  return (await r2.json()).id;
}

  S.tasks = m.tasks;
  S.settings = Object.assign({}, DEFAULTS.settings, m.settings || {});
  S.settingsEditedAt = m.settingsEditedAt;
  S.active = m.active;
  S.tombstones = m.tombstones;
}

async function syncNow() {
  if (gBusy || !syncOn()) return { ok: false, err: "kapalı" };
  gBusy = true;
  const before = S.sessions.length + S.tasks.length;
  try {
    const tok = await gAccessToken();
    let fileId = S.sync.fileId;
    if (!fileId) fileId = await driveFind(tok);
    const remote = fileId ? await driveRead(tok, fileId) : null;
    const R = (remote && Array.isArray(remote.sessions)) ? remote : { sessions: [], tasks: [], tombstones: [] };
    applyMerged(mergeStates(snapshot(), R));
    S.sync.fileId = await driveWrite(tok, fileId, snapshot());
    S.sync.lastAt = now(); S.sync.lastErr = null;
    save(); dilKur(); dilUygula(); applyTheme(); ctlKey = ""; render();
    return { ok: true, added: (S.sessions.length + S.tasks.length) - before };
  } catch (e) {
    S.sync.lastErr = String((e && e.message) || e);
    save();
    return { ok: false, err: S.sync.lastErr };
  } finally { gBusy = false; }
}

/* Sessiz otomatik eşitleme. Hata olursa ekranda hiçbir şey olmaz,
   uygulama yerel modda çalışmaya devam eder. */
function autoSync(force) {
  if (!syncOn() || !S.sync.auto) return;
  if (!force && now() - gLastTry < 120000) return;
  gLastTry = now();
  syncNow();
}

/* ---------------- senkron ekranı ---------------- */
function syncLabel() {
  if (!syncReady()) return soz("sy_yok");
  if (!S.sync.connected) return soz("sy_baglan");
  if (!S.sync.lastAt) return soz("sy_bagli");
  const d = now() - S.sync.lastAt;
  if (d < 12e4) return soz("sy_az_once");
  if (d < 36e5) return soz("sy_dk_once", { n: Math.round(d / 6e4) });
  if (d < 864e5) return soz("sy_sa_once", { n: Math.round(d / 36e5) });
  return soz("sy_gun_once", { n: Math.round(d / 864e5) });
}

/* Native tarafın döndürdüğü ham hatayı insan cümlesine çevirir.
   Ekranda kod, sayı, İngilizce görünmez. */
function yetkiHatasi(e) {
  const m = String((e && e.message) || e || "");
  if (m === "izin_gerekli") return soz("hata_izin");
  if (m === "reddedildi") return soz("hata_reddedildi");
  if (m === "baslatilamadi") return soz("hata_baslatilamadi");
  if (m === "yerel köprü yok") return soz("hata_yok");
  if (/DEVELOPER_ERROR|\b10\b/.test(m)) return soz("hata_developer");
  if (/NETWORK|\b7\b/.test(m)) return soz("hata_ag");
  return soz("hata_genel");
}
